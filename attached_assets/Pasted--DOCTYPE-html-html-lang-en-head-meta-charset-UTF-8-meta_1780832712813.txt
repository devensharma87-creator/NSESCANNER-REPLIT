    <script>
        // 1. Initialize Chart Properties
        const chartOptions = {
            layout: { 
                textColor: '#d1d4dc', 
                background: { type: 'solid', color: '#0d1117' } 
            },
            grid: {
                vertLines: { color: 'rgba(42, 46, 57, 0.4)' },
                horzLines: { color: 'rgba(42, 46, 57, 0.4)' },
            },
            crosshair: { 
                mode: LightweightCharts.CrosshairMode.Normal,
                vertLine: { color: '#758696', width: 1, style: 1, labelBackgroundColor: '#161b22' },
                horzLine: { color: '#758696', width: 1, style: 1, labelBackgroundColor: '#161b22' }
            },
            rightPriceScale: { borderColor: 'rgba(197, 203, 206, 0.4)' },
            timeScale: { borderColor: 'rgba(197, 203, 206, 0.4)', timeVisible: true },
        };

        const chartElement = document.getElementById('tvchart');
        const chart = LightweightCharts.createChart(chartElement, chartOptions);

        // 2. Add Candlestick Series
        const candleSeries = chart.addCandlestickSeries({
            upColor: '#238636', downColor: '#da3633', 
            borderVisible: false,
            wickUpColor: '#238636', wickDownColor: '#da3633',
        });

        // Mock Kite API Data Pipeline
        const candleData = [
            { time: 1717560000, open: 23300.50, high: 23350.00, low: 23280.00, close: 23340.20 },
            { time: 1717560900, open: 23340.20, high: 23410.00, low: 23320.00, close: 23380.50 },
            { time: 1717561800, open: 23380.50, high: 23450.00, low: 23370.00, close: 23437.95 },
            { time: 1717562700, open: 23437.95, high: 23460.00, low: 23390.00, close: 23410.00 },
        ];
        candleSeries.setData(candleData);

        // 3. Add Volume Series (Histogram at bottom)
        const volumeSeries = chart.addHistogramSeries({
            color: '#26a69a',
            priceFormat: { type: 'volume' },
            priceScaleId: '', // Set as an overlay
            scaleMargins: { top: 0.8, bottom: 0 }, // Pushes volume to bottom 20%
        });

        const volumeData = [
            { time: 1717560000, value: 4500000, color: 'rgba(35, 134, 54, 0.4)' },
            { time: 1717560900, value: 6200000, color: 'rgba(35, 134, 54, 0.4)' },
            { time: 1717561800, value: 5100000, color: 'rgba(35, 134, 54, 0.4)' },
            { time: 1717562700, value: 7200000, color: 'rgba(218, 54, 51, 0.4)' }, // Red for down candle
        ];
        volumeSeries.setData(volumeData);

        // 4. Add an EMA Overlay Example
        const emaSeries = chart.addLineSeries({
            color: '#58a6ff', lineWidth: 2, crosshairMarkerVisible: false,
        });
        emaSeries.setData([
            { time: 1717560000, value: 23310 }, { time: 1717560900, value: 23335 },
            { time: 1717561800, value: 23370 }, { time: 1717562700, value: 23395 },
        ]);

        // 5. Dynamic OHLC Legend Logic
        const legend = document.getElementById('legend');
        chart.subscribeCrosshairMove((param) => {
            if (param.point === undefined || !param.time || param.point.x < 0 || param.point.x > chartElement.clientWidth || param.point.y < 0 || param.point.y > chartElement.clientHeight) {
                legend.innerHTML = 'Hover over chart for data';
            } else {
                const data = param.seriesData.get(candleSeries);
                if (data) {
                    legend.innerHTML = `O <span style="color:#fff">${data.open}</span>  H <span style="color:#fff">${data.high}</span>  L <span style="color:#fff">${data.low}</span>  C <span style="color:#fff">${data.close}</span>`;
                }
            }
        });

        // Responsive resizing
        window.addEventListener('resize', () => {
            chart.resize(window.innerWidth, window.innerHeight - 100);
        });
    </script>
