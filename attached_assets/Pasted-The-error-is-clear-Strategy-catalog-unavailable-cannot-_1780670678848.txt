3. Frontend Fix: Load Strategy Catalog Properly

In your app.js, BacktestLab.jsx, or related frontend file, add this.

let backtestStrategies = [];
let selectedBacktestStrategies = new Set();

async function loadBacktestStrategies() {
  const container = document.getElementById("backtest-strategy-list");
  const warning = document.getElementById("backtest-strategy-warning");
  const runBtn = document.getElementById("run-backtest-btn");

  try {
    const res = await fetch("/api/backtest/strategies", {
      method: "GET",
      headers: { "Accept": "application/json" },
    });

    if (!res.ok) {
      throw new Error(`Strategy API failed with status ${res.status}`);
    }

    const data = await res.json();

    if (!data.ok || !Array.isArray(data.strategies) || data.strategies.length === 0) {
      throw new Error("No strategies returned from backend");
    }

    backtestStrategies = data.strategies.filter(s => s.enabled !== false);

    if (warning) {
      warning.style.display = "none";
      warning.textContent = "";
    }

    renderBacktestStrategyList();

    if (runBtn) {
      runBtn.disabled = selectedBacktestStrategies.size === 0;
    }

  } catch (err) {
    console.error("Backtest strategy catalog unavailable:", err);

    // Fallback so UI does not become useless
    backtestStrategies = getFallbackBacktestStrategies();

    if (warning) {
      warning.style.display = "block";
      warning.textContent =
        "Strategy catalog API unavailable. Using local fallback strategies. Backend must be fixed.";
    }

    renderBacktestStrategyList();

    if (runBtn) {
      runBtn.disabled = selectedBacktestStrategies.size === 0;
    }
  }
}

function renderBacktestStrategyList() {
  const container = document.getElementById("backtest-strategy-list");
  if (!container) return;

  container.innerHTML = "";

  backtestStrategies.forEach(strategy => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "strategy-chip";
    btn.dataset.strategyId = strategy.id;

    btn.innerHTML = `
      <span class="strategy-chip-title">${strategy.name}</span>
      <span class="strategy-chip-desc">${strategy.description || ""}</span>
    `;

    btn.onclick = () => {
      if (selectedBacktestStrategies.has(strategy.id)) {
        selectedBacktestStrategies.delete(strategy.id);
        btn.classList.remove("selected");
      } else {
        selectedBacktestStrategies.add(strategy.id);
        btn.classList.add("selected");
      }

      const runBtn = document.getElementById("run-backtest-btn");
      if (runBtn) {
        runBtn.disabled = selectedBacktestStrategies.size === 0;
      }
    };

    container.appendChild(btn);
  });
}

6. Fix Run Backtest Payload

Your run function must send selected strategies.

async function runStrategyBacktest() {
  const instrument = document.getElementById("backtest-instrument")?.value || "ALL";
  const startingCapital = Number(document.getElementById("backtest-capital")?.value || 1000000);
  const riskPerTrade = Number(document.getElementById("backtest-risk")?.value || 1);

  const strategies = Array.from(selectedBacktestStrategies);

  if (strategies.length === 0) {
    alert("Select at least one strategy to run.");
    return;
  }

  const payload = {
    mode: "STRATEGY_RESEARCH",
    instrument,
    starting_capital: startingCapital,
    risk_per_trade_pct: riskPerTrade,
    strategies,
    filters: {
      vwap_filter: true,
      ema_trend_filter: true,
      avoid_chop_zone: true,
      avoid_last_15_minutes: true,
      min_rr: 1.5,
      include_charges: true,
      include_slippage: true,
    },
  };

  try {
    const res = await fetch("/api/backtest/run", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Accept": "application/json",
      },
      body: JSON.stringify(payload),
    });

    if (!res.ok) {
      throw new Error(`Backtest failed with status ${res.status}`);
    }

    const data = await res.json();
    renderBacktestResults(data);

  } catch (err) {
    console.error("Backtest run failed:", err);
    alert("Backtest failed. Check backend logs.");
  }
}
7. Backend Run API Must Accept Strategies

Add validation like this.

from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any


class BacktestRunRequest(BaseModel):
    mode: str = "STRATEGY_RESEARCH"
    instrument: str = "ALL"
    starting_capital: float = Field(default=1000000, gt=0)
    risk_per_trade_pct: float = Field(default=1, gt=0)
    strategies: List[str]
    filters: Optional[Dict[str, Any]] = None


@router.post("/run")
def run_backtest(req: BacktestRunRequest):
    if not req.strategies:
        return {
            "ok": False,
            "error": "NO_STRATEGY_SELECTED",
            "message": "Select at least one strategy to run.",
        }

    allowed = {s["id"] for s in STRATEGY_CATALOG if s.get("enabled", True)}
    invalid = [s for s in req.strategies if s not in allowed]

    if invalid:
        return {
            "ok": False,
            "error": "INVALID_STRATEGY",
            "invalid_strategies": invalid,
        }

    # TODO: connect this to your actual backtest engine
    result = run_strategy_backtest_engine(
        mode=req.mode,
        instrument=req.instrument,
        starting_capital=req.starting_capital,
        risk_per_trade_pct=req.risk_per_trade_pct,
        strategies=req.strategies,
        filters=req.filters or {},
    )

    return {
        "ok": True,
        "mode": req.mode,
        "instrument": req.instrument,
        "strategies": req.strategies,
        "result": result,
    }
