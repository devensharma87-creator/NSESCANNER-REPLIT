-- Market Sentiment Score (0-100)
CREATE OR REPLACE FUNCTION market_sentiment_score()
RETURNS integer AS $$
DECLARE
    score integer := 50;
BEGIN
    -- FII/DII (30% weight)
    IF fii_net_3day_avg > 0 THEN score := score + 15;
    ELSIF fii_net_3day_avg < -2000 THEN score := score - 15;
    END IF;
    
    -- F&O OI (25% weight)
    IF client_oi_net > 0 AND fii_oi_net > 0 THEN score := score + 13;
    ELSIF client_oi_net < 0 AND fii_oi_net < 0 THEN score := score - 13;
    END IF;
    
    -- India VIX (20% weight)
    IF vix < 15 THEN score := score + 10;
    ELSIF vix > 22 THEN score := score - 10;
    END IF;
    
    -- PCR (15% weight)
    IF pcr > 1.2 THEN score := score + 8;
    ELSIF pcr < 0.8 THEN score := score - 8;
    END IF;
    
    -- NIFTY technical (10% weight)
    IF nifty > sma_20 AND nifty > sma_50 THEN score := score + 5;
    ELSIF nifty < sma_20 AND nifty < sma_50 THEN score := score - 5;
    END IF;
    
    RETURN score;
END;
$$ LANGUAGE plpgsql;

class KiteWebSocketClient:
    """
    Real-time market data WebSocket client for Zerodha Kite.
    Subscribes to ticks, stores in Redis/PostgreSQL, triggers signals.
    """
    
    def __init__(
        self,
        api_key: str,
        access_token: str,
        redis_client=None,
        db_conn=None,
        on_tick_callback: Optional[Callable] = None
    ):
        self.api_key = api_key
        self.access_token = access_token
        self.redis = redis_client
        self.db = db_conn
        self.callback = on_tick_callback
        
        self.kite = KiteConnect(api_key=api_key)
        self.kite.set_access_token(access_token)
        self.kws = None
        self.subscribed_tokens = set()
        
        # Token mappings (NSE, NFO, BSE)
        self.instrument_map = self._load_instrument_map()
        
    def _load_instrument_map(self) -> Dict:
        """Load instrument tokens from Kite (cached in DB)."""
        # In production, cache this in Redis/PostgreSQL
        instruments = self.kite.instruments()
        return {inst['tradingsymbol']: inst['instrument_token'] for inst in instruments}
    
    def _on_ticks(self, ws, ticks):
        """Called when ticks arrive."""
        for tick in ticks:
            try:
                data = {
                    'symbol': tick.get('tradingsymbol'),
                    'token': tick.get('instrument_token'),
                    'timestamp': datetime.fromtimestamp(tick['timestamp'] / 1000),
                    'ltp': tick['last_price'],
                    'volume': tick.get('volume', 0),
                    'oi': tick.get('oi', 0),
                    'open': tick.get('ohlc', {}).get('open'),
                    'high': tick.get('ohlc', {}).get('high'),
                    'low': tick.get('ohlc', {}).get('low'),
                    'close': tick.get('ohlc', {}).get('close'),
                    'change': tick.get('change', 0),
                    'buy_quantity': tick.get('buy_quantity', 0),
                    'sell_quantity': tick.get('sell_quantity', 0),
                }
                
                # Store in Redis (real-time cache, TTL 60 seconds)
                if self.redis:
                    self.redis.setex(
                        f"tick:{data['symbol']}",
                        60,
                        json.dumps(data, default=str)
                    )
                
                # Store in PostgreSQL (batched, not every tick)
                # Use a separate writer thread/process to avoid blocking
                
                # Trigger user callback
                if self.callback:
                    self.callback(data)
                    
                logger.debug(f"Tick: {data['symbol']} @ {data['ltp']}")
                
            except Exception as e:
                logger.error(f"Error processing tick: {e}")
    
    def _on_connection(self, ws, response):
        """WebSocket connected."""
        logger.info(f"WebSocket connected: {response}")
        # Resubscribe to all tokens
        if self.subscribed_tokens:
            self.kws.subscribe(self.subscribed_tokens)
            self.kws.set_mode(self.kws.MODE_FULL, self.subscribed_tokens)

        """WebSocket closed."""
        logger.warning(f"WebSocket closed: {close_status_code} - {close_msg}")
        # Auto-reconnect after 5 seconds
        time.sleep(5)
        self.connect()
    
    def _on_error(self, ws, error):
        logger.error(f"WebSocket error: {error}")
    
    def connect(self):
        """Establish WebSocket connection."""
        self.kws = KiteTicker(self.api_key, self.access_token)
        self.kws.on_ticks = self._on_ticks
        self.kws.on_connect = self._on_connection
        self.kws.on_close = self._on_close
        self.kws.on_error = self._on_error
        self.kws.connect()
    
    def subscribe(self, symbols: List[str]):
        """Subscribe to symbols (e.g., ['NIFTY', 'BANKNIFTY', 'RELIANCE'])."""
        tokens = []
        for sym in symbols:
            if sym in self.instrument_map:
                tokens.append(self.instrument_map[sym])
            else:
                logger.warning(f"Symbol not found: {sym}")
        
        if tokens and self.kws:
            self.kws.subscribe(tokens)
            self.kws.set_mode(self.kws.MODE_FULL, tokens)
            self.subscribed_tokens.update(tokens)
            logger.info(f"Subscribed to {len(tokens)} tokens")
    
    def unsubscribe(self, symbols: List[str]):
        """Unsubscribe from symbols."""
        tokens = [self.instrument_map[sym] for sym in symbols if sym in self.instrument_map]
        if tokens and self.kws:
            self.kws.unsubscribe(tokens)
            self.subscribed_tokens.difference_update(tokens)

import numpy as np
import pandas as pd
from scipy.stats import norm
from datetime import datetime, date
import requests

class OptionGreeksCalculator:
    """
    Calculate option Greeks (Delta, Gamma, Theta, Vega, Rho, IV)
    from NSE option chain data.
    """
    
    def __init__(self, risk_free_rate: float = 0.0675):  # 6.75% (Indian MCLR)
        self.r = risk_free_rate
    
    def black_scholes_greeks(
        self,
        S: float,          # Spot price
        K: float,          # Strike price
        T: float,          # Time to expiry in years
        sigma: float,      # Implied volatility (decimal)
        option_type: str   # 'call' or 'put'
    ) -> Dict[str, float]:
        """
        Calculate all Greeks using Black-Scholes model.
        
        Returns: dict with delta, gamma, theta, vega, rho
        """
        if T <= 0:
            T = 1e-6  # Avoid division by zero
        
        d1 = (np.log(S / K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        phi = norm.pdf  # probability density function
        Phi = norm.cdf  # cumulative distribution function
        
        if option_type.lower() == 'call':
            delta = Phi(d1)
            theta = (-S * phi(d1) * sigma / (2 * np.sqrt(T)) 
                     - self.r * K * np.exp(-self.r * T) * Phi(d2))
            rho = K * T * np.exp(-self.r * T) * Phi(d2)
        else:  # put
            delta = Phi(d1) - 1
            theta = (-S * phi(d1) * sigma / (2 * np.sqrt(T)) 
                     + self.r * K * np.exp(-self.r * T) * Phi(-d2))
            rho = -K * T * np.exp(-self.r * T) * Phi(-d2)
        
        gamma = phi(d1) / (S * sigma * np.sqrt(T))
        vega = S * phi(d1) * np.sqrt(T)  # Vega per 1% change in IV
        
        return {
            'delta': round(delta, 4),
            'gamma': round(gamma, 4),
            'theta': round(theta, 4),    # per day
            'vega': round(vega / 100, 4),  # per 1% IV change
            'rho': round(rho, 4),
            'iv': sigma
        }
    
    def implied_volatility(
        self,
        S: float,
        K: float,
        T: float,
        market_price: float,
        option_type: str,
        precision: float = 1e-6,
        max_iter: int = 100
    ) -> float:
        """
        Calculate implied volatility using Newton-Raphson method.
        """
        sigma_low = 0.01
        sigma_high = 2.0
        sigma_mid = 0.2
        
        for _ in range(max_iter):
            price_mid = self._black_scholes_price(S, K, T, sigma_mid, option_type)
            vega_mid = self._black_scholes_vega(S, K, T, sigma_mid, option_type)
            
            diff = price_mid - market_price
            
            if abs(diff) < precision:
                return sigma_mid
            
            if vega_mid == 0:
                break
            
            sigma_mid = sigma_mid - diff / vega_mid
            
            if sigma_mid < sigma_low:
                sigma_mid = sigma_low
            elif sigma_mid > sigma_high:
                sigma_mid = sigma_high
        
        # Fallback to binary search
        return self._binary_search_iv(S, K, T, market_price, option_type)
    
    def _black_scholes_price(self, S, K, T, sigma, option_type):
        d1 = (np.log(S / K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        if option_type.lower() == 'call':
            return S * norm.cdf(d1) - K * np.exp(-self.r * T) * norm.cdf(d2)
        else:
            return K * np.exp(-self.r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
    
    def _black_scholes_vega(self, S, K, T, sigma, option_type):
        d1 = (np.log(S / K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        return S * norm.pdf(d1) * np.sqrt(T)
    
    def _binary_search_iv(self, S, K, T, market_price, option_type):
        sigma_low, sigma_high = 0.01, 2.0
        for _ in range(100):
            sigma_mid = (sigma_low + sigma_high) / 2
            price = self._black_scholes_price(S, K, T, sigma_mid, option_type)
            if price < market_price:
                sigma_low = sigma_mid
            else:
                sigma_high = sigma_mid
        return (sigma_low + sigma_high) / 2


class NSEDataFetcher:
    """Fetch real-time option chain from NSE website."""
    
    BASE_URL = "https://www.nseindia.com"
    OPTION_CHAIN_URL = "https://www.nseindia.com/api/option-chain-indices?symbol={symbol}"
    
    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
            'Accept-Language': 'en-US,en;q=0.9',
            'Referer': 'https://www.nseindia.com/'
        })
    
    def _get_cookies(self):
        """Get initial cookies from NSE."""
        self.session.get(self.BASE_URL, timeout=10)
    
    def get_option_chain(self, symbol: str) -> Dict:
        """
        Fetch option chain for NIFTY, BANKNIFTY, FINNIFTY, etc.
        Returns raw JSON response.
        """
        self._get_cookies()
        url = self.OPTION_CHAIN_URL.format(symbol=symbol)
        response = self.session.get(url, timeout=15)
        if response.status_code == 200:
            return response.json()
        else:
            raise Exception(f"NSE API error: {response.status_code}")
    
    def extract_greeks_from_chain(self, symbol: str, spot_price: float) -> pd.DataFrame:
        """
        Extract all strikes, LTP, and compute Greeks for each.
        """
        data = self.get_option_chain(symbol)
        records = []
        calculator = OptionGreeksCalculator()
        
        expiry_dates = [exp['expiryDate'] for exp in data['records']['expiryDates']]
        # Use nearest expiry
        nearest_expiry = expiry_dates[0]
        expiry_date = datetime.strptime(nearest_expiry, '%d-%b-%Y')
        T = (expiry_date - datetime.now()).days / 365.0
        
        for strike_data in data['records']['data']:
            strike = strike_data['strikePrice']
            
            # CE data
            ce = strike_data.get('CE', {})
            if ce:
                premium = ce.get('lastPrice', 0)
                if premium > 0:
                    iv = calculator.implied_volatility(spot_price, strike, T, premium, 'call')
                    greeks = calculator.black_scholes_greeks(spot_price, strike, T, iv, 'call')
                    records.append({
                        'symbol': symbol,
                        'expiry': nearest_expiry,
                        'strike': strike,
                        'option_type': 'CE',
                        'ltp': premium,
                        'oi': ce.get('openInterest', 0),
                        'change_oi': ce.get('changeinOpenInterest', 0),
                        'volume': ce.get('totalTradedVolume', 0),
                        'iv': iv,
                        **greeks
                    })
            
            # PE data
            pe = strike_data.get('PE', {})
            if pe:
                premium = pe.get('lastPrice', 0)
                if premium > 0:
                    iv = calculator.implied_volatility(spot_price, strike, T, premium, 'put')
                    greeks = calculator.black_scholes_greeks(spot_price, strike, T, iv, 'put')
                    records.append({
                        'symbol': symbol,
                        'expiry': nearest_expiry,
                        'strike': strike,
                        'option_type': 'PE',
                        'ltp': premium,
                        'oi': pe.get('openInterest', 0),
                        'change_oi': pe.get('changeinOpenInterest', 0),
                        'volume': pe.get('totalTradedVolume', 0),
                        'iv': iv,
                        **greeks
                    })
        
        return pd.DataFrame(records)

    def backtest_signal(
        self,
        signal_df: pd.DataFrame,
        candles_df: pd.DataFrame,
        premium_df: pd.DataFrame,
        risk_per_trade: float = 0.02,    # 2% of capital
        capital: float = 1000000,
        slippage_pct: float = 0.001,
        brokerage_per_lot: float = 40
    ) -> BacktestResult:
        """
        Run backtest on a set of signals.
        
        signal_df must have columns: timestamp, signal_type (BUY/SELL), entry_price, stop_loss, target1, target2
        candles_df: OHLC data for entry/exit logic
        premium_df: option premiums for P&L calculation
        """
        trades = []
        equity_curve = [capital]
        
        for idx, signal in signal_df.iterrows():
            entry_time = signal['timestamp']
            entry_price = signal['entry_price']
            stop = signal['stop_loss']
            target1 = signal['target1']
            target2 = signal['target2']
            direction = signal['signal_type']
            
            # Find subsequent bars to check exit conditions
            future_candles = candles_df[candles_df['ts'] > entry_time].copy()
            future_premiums = premium_df[premium_df['ts'] > entry_time].copy()
            
            exit_time = None
            exit_price = None
            exit_reason = None
            
            # Simulate price movement minute-by-minute (using close prices)
            max_favorable = 0
            max_adverse = 0
            
            for _, candle in future_candles.iterrows():
                current_time = candle['ts']
                # Get premium at this time (use last known)
                premium_row = future_premiums[future_premiums['ts'] <= current_time].iloc[-1] if not future_premiums.empty else None
                current_premium = premium_row['premium'] if premium_row is not None else entry_price
                
                # Calculate P&L
                if direction == 'BUY':
                    pnl = (current_premium - entry_price) / entry_price
                else:  # SELL (short)
                    pnl = (entry_price - current_premium) / entry_price
                
                max_favorable = max(max_favorable, pnl)
                max_adverse = min(max_adverse, pnl)
                
                # Check stop loss
                if direction == 'BUY' and current_premium <= stop:
                    exit_time = current_time
                    exit_price = stop
                    exit_reason = 'STOPPED'
                    break
                elif direction == 'SELL' and current_premium >= stop:
                    exit_time = current_time
                    exit_price = stop
                    exit_reason = 'STOPPED'
                    break
                
                # Check targets
                if direction == 'BUY':
                    if current_premium >= target2:
                        exit_time = current_time
                        exit_price = target2
                        exit_reason = 'TARGET2_HIT'
                        break
                    elif current_premium >= target1:
                        # Partial booking logic: book 50% at T1, let runner run
                        # For simplicity, we'll exit fully at T1 if T2 not reached
                        exit_time = current_time
                        exit_price = target1
                        exit_reason = 'TARGET1_HIT'
                        break
                else:  # SELL
                    if current_premium <= target2:
                        exit_time = current_time
                        exit_price = target2
                        exit_reason = 'TARGET2_HIT'
                        break
                    elif current_premium <= target1:
                        exit_time = current_time
                        exit_price = target1
                        exit_reason = 'TARGET1_HIT'
                        break
            
            # If no exit, expire at end
            if exit_time is None:
                exit_time = future_candles.iloc[-1]['ts']
                exit_price = future_premiums.iloc[-1]['premium'] if not future_premiums.empty else entry_price
                exit_reason = 'EXPIRED'
            
            # Calculate trade P&L
            if direction == 'BUY':
                pnl_pct = (exit_price - entry_price) / entry_price
            else:
                pnl_pct = (entry_price - exit_price) / entry_price
            
            # Adjust for slippage and brokerage
            pnl_pct -= slippage_pct * 2  # entry + exit
            trade_pnl = capital * risk_per_trade * pnl_pct - brokerage_per_lot
            
            trades.append({
                'entry_time': entry_time,
                'exit_time': exit_time,
                'direction': direction,
                'entry_price': entry_price,
                'exit_price': exit_price,
                'exit_reason': exit_reason,
                'pnl': trade_pnl,
                'pnl_pct': pnl_pct,
                'max_favorable': max_favorable,
                'max_adverse': max_adverse
            })
            
            # Update equity curve
            equity_curve.append(equity_curve[-1] + trade_pnl)
        
        # Compute metrics
        df_trades = pd.DataFrame(trades)
        winning = df_trades[df_trades['pnl'] > 0]
        losing = df_trades[df_trades['pnl'] < 0]
        
        total_trades = len(df_trades)
        winning_trades = len(winning)
        losing_trades = len(losing)
        win_rate = winning_trades / total_trades if total_trades > 0 else 0
        
        total_pnl = df_trades['pnl'].sum()
        avg_win = winning['pnl'].mean() if winning_trades > 0 else 0
        avg_loss = losing['pnl'].mean() if losing_trades > 0 else 0
        profit_factor = abs(winning['pnl'].sum() / losing['pnl'].sum()) if losing['pnl'].sum() != 0 else np.inf
        
        # Max drawdown
        equity_series = pd.Series(equity_curve)
        rolling_max = equity_series.expanding().max()
        drawdown = (equity_series - rolling_max) / rolling_max
        max_drawdown = drawdown.min()
        
        # Sharpe ratio (daily returns approximated)
        returns = equity_series.pct_change().dropna()
        sharpe_ratio = (returns.mean() / returns.std()) * np.sqrt(252) if returns.std() != 0 else 0
        
        return BacktestResult(
            total_trades=total_trades,

            losing_trades=losing_trades,

            total_pnl=total_pnl,
            avg_win=avg_win,
            avg_loss=avg_loss,
            profit_factor=profit_factor,
            max_drawdown=max_drawdown,
            sharpe_ratio=sharpe_ratio,
            trade_log=trades
        )

from typing import Dict, List, Callable, Optional

from kiteconnect import KiteTicker, KiteConnect

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class KiteWebSocketClient:
    """
    Real-time market data WebSocket client for Zerodha Kite.
    Subscribes to ticks, stores in Redis/PostgreSQL, triggers signals.
    """
    
    def __init__(
        self,
        api_key: str,
        access_token: str,
        redis_client=None,
        db_conn=None,
        on_tick_callback: Optional[Callable] = None
    ):
        self.api_key = api_key
        self.access_token = access_token
        self.redis = redis_client
        self.db = db_conn
        self.callback = on_tick_callback
        
        self.kite = KiteConnect(api_key=api_key)
        self.kite.set_access_token(access_token)
        self.kws = None
        self.subscribed_tokens = set()
        
        # Token mappings (NSE, NFO, BSE)
        self.instrument_map = self._load_instrument_map()
        
    def _load_instrument_map(self) -> Dict:
        """Load instrument tokens from Kite (cached in DB)."""
        # In production, cache this in Redis/PostgreSQL
        instruments = self.kite.instruments()
        return {inst['tradingsymbol']: inst['instrument_token'] for inst in instruments}
    
    def _on_ticks(self, ws, ticks):
        """Called when ticks arrive."""
        for tick in ticks:
            try:
                data = {
                    'symbol': tick.get('tradingsymbol'),
                    'token': tick.get('instrument_token'),

                    )

                    self.callback(data)
                    
                logger.debug(f"Tick: {data['symbol']} @ {data['ltp']}")
                
            except Exception as e:
                logger.error(f"Error processing tick: {e}")
    
    def _on_connection(self, ws, response):
        """WebSocket connected."""
        logger.info(f"WebSocket connected: {response}")
        # Resubscribe to all tokens
        if self.subscribed_tokens:
            self.kws.subscribe(self.subscribed_tokens)
            self.kws.set_mode(self.kws.MODE_FULL, self.subscribed_tokens)
    
    def _on_close(self, ws, close_status_code, close_msg):
        """WebSocket closed."""
        logger.warning(f"WebSocket closed: {close_status_code} - {close_msg}")
        # Auto-reconnect after 5 seconds
        time.sleep(5)
        self.connect()
    
    def _on_error(self, ws, error):
        logger.error(f"WebSocket error: {error}")
    
    def connect(self):
        """Establish WebSocket connection."""
        self.kws = KiteTicker(self.api_key, self.access_token)
        self.kws.on_ticks = self._on_ticks
        self.kws.on_connect = self._on_connection
        self.kws.on_close = self._on_close
        self.kws.on_error = self._on_error
        self.kws.connect()
    
    def subscribe(self, symbols: List[str]):
        """Subscribe to symbols (e.g., ['NIFTY', 'BANKNIFTY', 'RELIANCE'])."""
        tokens = []
        for sym in symbols:
            if sym in self.instrument_map:
                tokens.append(self.instrument_map[sym])
            else:
                logger.warning(f"Symbol not found: {sym}")
        
        if tokens and self.kws:
            self.kws.subscribe(tokens)
            self.kws.set_mode(self.kws.MODE_FULL, tokens)
            self.subscribed_tokens.update(tokens)
            logger.info(f"Subscribed to {len(tokens)} tokens")
    
    def unsubscribe(self, symbols: List[str]):
        """Unsubscribe from symbols."""
        tokens = [self.instrument_map[sym] for sym in symbols if sym in self.instrument_map]
        if tokens and self.kws:
            self.kws.unsubscribe(tokens)
            self.subscribed_tokens.difference_update(tokens)

import numpy as np
import pandas as pd
from scipy.stats import norm
from datetime import datetime, date

class OptionGreeksCalculator:
    """
    Calculate option Greeks (Delta, Gamma, Theta, Vega, Rho, IV)
    from NSE option chain data.
    """
    
    def __init__(self, risk_free_rate: float = 0.0675):  # 6.75% (Indian MCLR)
        self.r = risk_free_rate
    
    def black_scholes_greeks(
        self,
        S: float,          # Spot price
        K: float,          # Strike price
        T: float,          # Time to expiry in years
        sigma: float,      # Implied volatility (decimal)
        option_type: str   # 'call' or 'put'
    ) -> Dict[str, float]:
        """
        Calculate all Greeks using Black-Scholes model.
        
        Returns: dict with delta, gamma, theta, vega, rho
        """
        if T <= 0:
            T = 1e-6  # Avoid division by zero
        
        d1 = (np.log(S / K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        phi = norm.pdf  # probability density function
        Phi = norm.cdf  # cumulative distribution function
        
        if option_type.lower() == 'call':
            delta = Phi(d1)
            theta = (-S * phi(d1) * sigma / (2 * np.sqrt(T)) 
                     - self.r * K * np.exp(-self.r * T) * Phi(d2))
            rho = K * T * np.exp(-self.r * T) * Phi(d2)
        else:  # put
            delta = Phi(d1) - 1
            theta = (-S * phi(d1) * sigma / (2 * np.sqrt(T)) 
                     + self.r * K * np.exp(-self.r * T) * Phi(-d2))
            rho = -K * T * np.exp(-self.r * T) * Phi(-d2)
        
        gamma = phi(d1) / (S * sigma * np.sqrt(T))
        vega = S * phi(d1) * np.sqrt(T)  # Vega per 1% change in IV
        
        return {
            'delta': round(delta, 4),
            'gamma': round(gamma, 4),
            'theta': round(theta, 4),    # per day
            'vega': round(vega / 100, 4),  # per 1% IV change
            'rho': round(rho, 4),
            'iv': sigma
        }
    
    def implied_volatility(
        self,
        S: float,
        K: float,
        T: float,
        market_price: float,
        option_type: str,
        precision: float = 1e-6,
        max_iter: int = 100
    ) -> float:
        """
        Calculate implied volatility using Newton-Raphson method.
        """
        sigma_low = 0.01
        sigma_high = 2.0
        sigma_mid = 0.2
        
        for _ in range(max_iter):
            price_mid = self._black_scholes_price(S, K, T, sigma_mid, option_type)
            vega_mid = self._black_scholes_vega(S, K, T, sigma_mid, option_type)
            
            diff = price_mid - market_price
            
            if abs(diff) < precision:
                return sigma_mid
            
            if vega_mid == 0:
                break
            
            sigma_mid = sigma_mid - diff / vega_mid
            
            if sigma_mid < sigma_low:
                sigma_mid = sigma_low
            elif sigma_mid > sigma_high:
                sigma_mid = sigma_high
        
        # Fallback to binary search
        return self._binary_search_iv(S, K, T, market_price, option_type)
    
    def _black_scholes_price(self, S, K, T, sigma, option_type):
        d1 = (np.log(S / K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        if option_type.lower() == 'call':
            return S * norm.cdf(d1) - K * np.exp(-self.r * T) * norm.cdf(d2)
        else:
            return K * np.exp(-self.r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
    
    def _black_scholes_vega(self, S, K, T, sigma, option_type):
        d1 = (np.log(S / K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        return S * norm.pdf(d1) * np.sqrt(T)
    
    def _binary_search_iv(self, S, K, T, market_price, option_type):
        sigma_low, sigma_high = 0.01, 2.0
        for _ in range(100):
            sigma_mid = (sigma_low + sigma_high) / 2
            price = self._black_scholes_price(S, K, T, sigma_mid, option_type)
            if price < market_price:
                sigma_low = sigma_mid
            else:
                sigma_high = sigma_mid
        return (sigma_low + sigma_high) / 2


class NSEDataFetcher:
    """Fetch real-time option chain from NSE website."""
    
    BASE_URL = "https://www.nseindia.com"
    OPTION_CHAIN_URL = "https://www.nseindia.com/api/option-chain-indices?symbol={symbol}"
    
    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
            'Accept-Language': 'en-US,en;q=0.9',
            'Referer': 'https://www.nseindia.com/'
        })
    
    def _get_cookies(self):
        """Get initial cookies from NSE."""
        self.session.get(self.BASE_URL, timeout=10)
    
    def get_option_chain(self, symbol: str) -> Dict:
        """
        Fetch option chain for NIFTY, BANKNIFTY, FINNIFTY, etc.
        Returns raw JSON response.
        """
        self._get_cookies()
        url = self.OPTION_CHAIN_URL.format(symbol=symbol)
        response = self.session.get(url, timeout=15)
        if response.status_code == 200:
            return response.json()
        else:
            raise Exception(f"NSE API error: {response.status_code}")
    
    def extract_greeks_from_chain(self, symbol: str, spot_price: float) -> pd.DataFrame:
        """
        Extract all strikes, LTP, and compute Greeks for each.
        """
        data = self.get_option_chain(symbol)
        records = []
        calculator = OptionGreeksCalculator()
        
        expiry_dates = [exp['expiryDate'] for exp in data['records']['expiryDates']]
        # Use nearest expiry
        nearest_expiry = expiry_dates[0]
        expiry_date = datetime.strptime(nearest_expiry, '%d-%b-%Y')
        T = (expiry_date - datetime.now()).days / 365.0
        
        for strike_data in data['records']['data']:
            strike = strike_data['strikePrice']
            
            # CE data
            ce = strike_data.get('CE', {})
            if ce:
                premium = ce.get('lastPrice', 0)
                if premium > 0:
                    iv = calculator.implied_volatility(spot_price, strike, T, premium, 'call')
                    greeks = calculator.black_scholes_greeks(spot_price, strike, T, iv, 'call')
                    records.append({
                        'symbol': symbol,
                        'expiry': nearest_expiry,
                        'strike': strike,
                        'option_type': 'CE',
                        'ltp': premium,
                        'oi': ce.get('openInterest', 0),
                        'change_oi': ce.get('changeinOpenInterest', 0),
                        'volume': ce.get('totalTradedVolume', 0),
                        'iv': iv,
                        **greeks
                    })
            
            # PE data
            pe = strike_data.get('PE', {})
            if pe:
                premium = pe.get('lastPrice', 0)
                if premium > 0:
                    iv = calculator.implied_volatility(spot_price, strike, T, premium, 'put')
                    greeks = calculator.black_scholes_greeks(spot_price, strike, T, iv, 'put')
                    records.append({
                        'symbol': symbol,
                        'expiry': nearest_expiry,
                        'strike': strike,
                        'option_type': 'PE',
                        'ltp': premium,
                        'oi': pe.get('openInterest', 0),
                        'change_oi': pe.get('changeinOpenInterest', 0),
                        'volume': pe.get('totalTradedVolume', 0),
                        'iv': iv,
                        **greeks
                    })
        
        return pd.DataFrame(records)

    def backtest_signal(
        self,
        signal_df: pd.DataFrame,
        candles_df: pd.DataFrame,
        premium_df: pd.DataFrame,
        risk_per_trade: float = 0.02,    # 2% of capital
        capital: float = 1000000,
        slippage_pct: float = 0.001,
        brokerage_per_lot: float = 40
    ) -> BacktestResult:
        """
        Run backtest on a set of signals.
        
        signal_df must have columns: timestamp, signal_type (BUY/SELL), entry_price, stop_loss, target1, target2
        candles_df: OHLC data for entry/exit logic
        premium_df: option premiums for P&L calculation
        """
        trades = []
        equity_curve = [capital]
        
        for idx, signal in signal_df.iterrows():
            entry_time = signal['timestamp']
            entry_price = signal['entry_price']
            stop = signal['stop_loss']
            target1 = signal['target1']
            target2 = signal['target2']
            direction = signal['signal_type']
            
            # Find subsequent bars to check exit conditions
            future_candles = candles_df[candles_df['ts'] > entry_time].copy()
            future_premiums = premium_df[premium_df['ts'] > entry_time].copy()
            
            exit_time = None
            exit_price = None
            exit_reason = None
            
            # Simulate price movement minute-by-minute (using close prices)
            max_favorable = 0
            max_adverse = 0
            
            for _, candle in future_candles.iterrows():
                current_time = candle['ts']
                # Get premium at this time (use last known)
                premium_row = future_premiums[future_premiums['ts'] <= current_time].iloc[-1] if not future_premiums.empty else None
                current_premium = premium_row['premium'] if premium_row is not None else entry_price
                
                # Calculate P&L
                if direction == 'BUY':
                    pnl = (current_premium - entry_price) / entry_price
                else:  # SELL (short)
                    pnl = (entry_price - current_premium) / entry_price
                
                max_favorable = max(max_favorable, pnl)
                max_adverse = min(max_adverse, pnl)
                
                # Check stop loss
                if direction == 'BUY' and current_premium <= stop:
                    exit_time = current_time
                    exit_price = stop
                    exit_reason = 'STOPPED'
                    break
                elif direction == 'SELL' and current_premium >= stop:
                    exit_time = current_time
                    exit_price = stop
                    exit_reason = 'STOPPED'
                    break
                
                # Check targets
                if direction == 'BUY':
                    if current_premium >= target2:
                        exit_time = current_time
                        exit_price = target2
                        exit_reason = 'TARGET2_HIT'
                        break
                    elif current_premium >= target1:
                        # Partial booking logic: book 50% at T1, let runner run
                        # For simplicity, we'll exit fully at T1 if T2 not reached
                        exit_time = current_time
                        exit_price = target1
                        exit_reason = 'TARGET1_HIT'
                        break
                else:  # SELL
                    if current_premium <= target2:
                        exit_time = current_time
                        exit_price = target2
                        exit_reason = 'TARGET2_HIT'
                        break
                    elif current_premium <= target1:
                        exit_time = current_time
                        exit_price = target1
                        exit_reason = 'TARGET1_HIT'
                        break
            
            # If no exit, expire at end
            if exit_time is None:
                exit_time = future_candles.iloc[-1]['ts']
                exit_price = future_premiums.iloc[-1]['premium'] if not future_premiums.empty else entry_price
                exit_reason = 'EXPIRED'
            
            # Calculate trade P&L
            if direction == 'BUY':
                pnl_pct = (exit_price - entry_price) / entry_price
            else:
                pnl_pct = (entry_price - exit_price) / entry_price
            
            # Adjust for slippage and brokerage
            pnl_pct -= slippage_pct * 2  # entry + exit
            trade_pnl = capital * risk_per_trade * pnl_pct - brokerage_per_lot
            
            trades.append({
                'entry_time': entry_time,
                'exit_time': exit_time,
                'direction': direction,
                'entry_price': entry_price,
                'exit_price': exit_price,
                'exit_reason': exit_reason,
                'pnl': trade_pnl,
                'pnl_pct': pnl_pct,
                'max_favorable': max_favorable,
                'max_adverse': max_adverse
            })
            
            # Update equity curve
            equity_curve.append(equity_curve[-1] + trade_pnl)
        
        # Compute metrics
        df_trades = pd.DataFrame(trades)
        winning = df_trades[df_trades['pnl'] > 0]
        losing = df_trades[df_trades['pnl'] < 0]
        
        total_trades = len(df_trades)
        winning_trades = len(winning)
        losing_trades = len(losing)
        win_rate = winning_trades / total_trades if total_trades > 0 else 0
        
        total_pnl = df_trades['pnl'].sum()
        avg_win = winning['pnl'].mean() if winning_trades > 0 else 0
        avg_loss = losing['pnl'].mean() if losing_trades > 0 else 0
        profit_factor = abs(winning['pnl'].sum() / losing['pnl'].sum()) if losing['pnl'].sum() != 0 else np.inf
        
        # Max drawdown
        equity_series = pd.Series(equity_curve)
        rolling_max = equity_series.expanding().max()
        drawdown = (equity_series - rolling_max) / rolling_max
        max_drawdown = drawdown.min()
        
        # Sharpe ratio (daily returns approximated)
        returns = equity_series.pct_change().dropna()
        sharpe_ratio = (returns.mean() / returns.std()) * np.sqrt(252) if returns.std() != 0 else 0
        
        return BacktestResult(
            total_trades=total_trades,
            winning_trades=winning_trades,
            losing_trades=losing_trades,
            win_rate=win_rate,
            total_pnl=total_pnl,
            avg_win=avg_win,
            avg_loss=avg_loss,
            profit_factor=profit_factor,
            max_drawdown=max_drawdown,
            sharpe_ratio=sharpe_ratio,
            trade_log=trades
        )

          'MACD@tv-basicstudies',
          'BB@tv-basicstudies'
        ],
        width: '100%',
        height: 500,
      });
    };
    document.head.appendChild(script);

    return () => {
      if (script.parentNode) script.parentNode.removeChild(script);
