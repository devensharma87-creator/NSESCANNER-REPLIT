        array.clear(vap)

            array.push(vap, 0.0)
        for i = 0 to vpLB - 1
            int ri = math.min(int((close[i] - loR) / rH), vpRows - 1)

                array.set(vap, ri, array.get(vap, ri) + volume[i])

        for r = 0 to vpRows - 1
            if array.get(vap, r) > maxV
                maxV := array.get(vap, r)
                pIdx := r

        for r = 0 to vpRows - 1
            totV += array.get(vap, r)

            float aH = vHi < vpRows - 1 ? array.get(vap, vHi + 1) : 0.0

                vHi += 1
                vaV += aH
            else if vLo > 0
                vLo -= 1
                vaV += aL
            else
                vHi += 1
                vaV += aH

        valLvl := loR + vLo * rH
