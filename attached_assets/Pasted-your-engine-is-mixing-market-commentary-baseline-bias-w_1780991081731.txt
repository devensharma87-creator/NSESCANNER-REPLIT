Fresh PUT should be blocked if market is in recovery mode.

Add this rule:

def is_recovery_mode(candles, spot, vwap, ema9, ema21, rsi, day_low):
    last = candles[-1]
    prev = candles[-4:-1]

    bounced_from_low = (spot - day_low) >= 0.35 * current_atr
    higher_lows = candles[-1].low > candles[-2].low > candles[-3].low
    rsi_reclaim = rsi.current > rsi.prev and rsi.current >= 45
    reclaiming_vwap = spot > ema9 or spot >= vwap * 0.998

    return bounced_from_low and (higher_lows or rsi_reclaim) and reclaiming_vwap
