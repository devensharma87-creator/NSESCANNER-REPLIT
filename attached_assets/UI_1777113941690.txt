📊 5. Market Cards (Key Upgrade)
const MarketCard = ({ title, price, change }) => {
  const isPositive = change > 0;

  return (
    <div className="bg-[#111827] border border-[rgba(255,255,255,0.06)] rounded-xl p-4">
      <div className="text-gray-400 text-sm">{title}</div>

      <div className="flex items-center justify-between mt-2">
        <div className="text-2xl font-semibold number text-white">
          {price}
        </div>

        <div className={isPositive ? "text-green-400" : "text-red-400"}>
          {change}%
        </div>
      </div>

      <div className="text-xs text-gray-500 mt-2">
        Open · High · Low
      </div>
    </div>
  );
};
