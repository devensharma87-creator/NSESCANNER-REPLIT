```ts
export type FnoPaperRiskGuardReason =
  | "SENSEX_DISABLED_BY_REPLAY_DIAGNOSTICS"
  | "LOW_ENTRY_PREMIUM"
  | "NEAR_EXPIRY_THETA_RISK"
  | "SAME_STRIKE_DIRECTION_STOP_COOLDOWN"
  | "BAD_TIME_WINDOW_SHADOW_ONLY"
  | "HIGH_COST_TO_EDGE_RATIO_SHADOW_ONLY";

export interface FnoPaperRiskGuardInput {
  underlying: "NIFTY" | "BANKNIFTY" | "SENSEX";
  direction: "BULLISH" | "BEARISH";
  optionType: "CALL" | "PUT";
  strike: number;
  entryPremium: number | null;
  entryTime: string;
  expiry: string | null;
  pricingMode?: string | null;
  setupKey?: string | null;
  setupName?: string | null;
  candidateTier?: string | null;
  premiumTrusted?: boolean | null;
  expectedGrossEdge?: number | null;
  estimatedCosts?: number | null;
}

export interface RecentStoppedTrade {
  underlying: string;
  direction: string;
  optionType: string;
  strike: number;
  exitTime: string;
  exitReason: string;
}

export interface FnoPaperRiskGuardDecision {
  allowed: boolean;
  severity: "info" | "warn" | "block";
  reasons: FnoPaperRiskGuardReason[];
  explanation: string[];
  metrics: {
    dteCalendarDays: number | null;
    entryPremium: number | null;
    minPremiumRequired: number | null;
    minutesSinceSameStrikeStop: number | null;
    costToEdgeRatio: number | null;
  };
}
```

```ts
evaluateFnoPaperRiskGuards(
```

Critical rule:

DTE/theta guard must NOT block every DTE ≤ 5 trade blindly.

It should block only when:

* DTE <= configured max DTE
  AND
* entry premium is below that underlying’s minimum premium threshold
  OR
* option is materially OTM if moneyness is available
  OR
* expected gross edge is lower than estimated costs if those fields exist

Reason:

The diagnostic damage came from near-expiry low-premium theta traps, not from every near-expiry trade.
