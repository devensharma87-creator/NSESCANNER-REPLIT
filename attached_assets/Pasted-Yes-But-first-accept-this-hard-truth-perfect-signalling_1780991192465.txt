def classify_regime(f):
    bullish_stack = f.close > f.vwap and f.ema9 > f.ema21 > f.ema50
    bearish_stack = f.close < f.vwap and f.ema9 < f.ema21 < f.ema50

    higher_structure = f.hh_count >= 2 and f.hl_count >= 2
    lower_structure = f.lh_count >= 2 and f.ll_count >= 2

    trend_strength = f.adx >= 18 and f.adx_slope >= 0

    if bullish_stack and higher_structure and f.rsi > 52 and trend_strength:
        return "TREND_UP"

    if bearish_stack and lower_structure and f.rsi < 48 and trend_strength:
        return "TREND_DOWN"

    if f.recovering_from_day_low and f.rsi_slope > 0 and f.close > f.ema9:
        return "RECOVERY"

    if f.adx < 16 and abs(f.close - f.vwap) < 0.25 * f.atr:
        return "RANGE_CHOP"

    return "NO_TRADE"

Step 4 — Recovery veto

This is mandatory.

def recovery_mode_veto(direction, f):
    recovery = (
        f.close > f.day_low + 0.35 * f.atr
        and f.rsi > f.rsi_prev
        and f.rsi >= 45
        and (f.close > f.ema9 or f.close > f.vwap - 0.20 * f.atr)
        and f.hl_count_last_3 >= 2
    )

    if direction == "PUT" and recovery:
        return reject("RECOVERY_MODE_VETO")

    return pass_gate()

Meaning: after market bounces from intraday low, RSI rises, and candles start making higher lows, no fresh PUT unless a new breakdown is confirmed.

Late CALL after vertical move should also be blocked.

def chase_risk_veto(direction, f):
    extended_from_vwap = abs(f.close - f.vwap) > 1.20 * f.atr
    extended_from_ema9 = abs(f.close - f.ema9) > 0.75 * f.atr
    rsi_extreme = f.rsi > 70 if direction == "CALL" else f.rsi < 30
    vertical_candles = f.same_color_large_candles_last_3 >= 2

    if extended_from_vwap and extended_from_ema9 and rsi_extreme and vertical_candles:
        return reject("CHASE_RISK_WAIT_FOR_PULLBACK")

    return pass_gate()

This protects you from buying CALL at the top after the engine missed the clean entry.

def anti_flip_gate(last_signal, new_signal, f):
    if not last_signal:
        return pass_gate()

    if last_signal.direction == new_signal.direction:
        return pass_gate()

    confirmed_flip = (
        f.two_candle_close_across_vwap
        and f.rsi_crossed_50_in_new_direction
        and f.ema9_slope_changed
        and f.previous_setup_invalidated
    )

    if not confirmed_flip:
        return reject("ANTI_FLIP_CONFIRMATION_REQUIRED")

    return pass_gate()

After two losses: no more signals for that index

def build_trade_plan(direction, f):
    if direction == "CALL":
        entry = f.trigger_high + f.tick_size
        stop = min(f.swing_low, f.ema21, f.vwap) - 0.15 * f.atr
    else:
        entry = f.trigger_low - f.tick_size
        stop = max(f.swing_high, f.ema21, f.vwap) + 0.15 * f.atr

    risk = abs(entry - stop)
    t1 = entry + 1.0 * risk if direction == "CALL" else entry - 1.0 * risk
    t2 = entry + 1.8 * risk if direction == "CALL" else entry - 1.8 * risk

    if risk > 0.65 * f.atr:
        return reject("STOP_TOO_WIDE")

    if risk < 0.15 * f.atr:
        return reject("STOP_TOO_TIGHT_NOISE_RISK")

    if expected_rr(entry, stop, t2) < 1.5:
        return reject("MIN_RR_NOT_MET")

    return plan(entry, stop, t1, t2)

Use spot levels for signal. Option premium is only execution mapping.

Step 10 — Final emission gate

Only emit a signal when all pass:

def emit_signal(index, f):
    quality = data_quality_gate(f)
    if not quality.ok:
        return no_trade(quality.reason)

    regime = classify_regime(f)
    if regime in ["RANGE_CHOP", "VOLATILE_NOISE", "NO_TRADE"]:
        return no_trade(regime)

    direction, direction_score = score_direction(f)

    if direction_score < 65:
        return info_only("BIAS_ONLY_NOT_TRADEABLE")

    setup = detect_setup(direction, f)
    if not setup:
        return watchlist("NO_TRIGGER_YET")

    for gate in [
        recovery_mode_veto,
        chase_risk_veto,
        anti_flip_gate,
        option_chain_gate,
        risk_gate,
        time_gate,
        circuit_gate,
    ]:
        result = gate(direction, f)
        if not result.ok:
            return watchlist_or_no_trade(result.reason)

    if direction_score >= 75 and setup.trigger_confirmed and setup.rr >= 1.5:
        return tradeable_signal(index, direction, setup)

    return watchlist("WAITING_FOR_CONFIRMATION")
